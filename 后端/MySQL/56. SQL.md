```sql
show databases; # 显示所有数据库

create database  if not exists db_workshop character set utf8mb4 collate utf8mb4_general_ci; # 创建数据库 db_workshop，字符集 utf8mb4，排序方式 utf8mb4_general_ci

select database(); # 显示当前使用的数据库

show create database db_workshop; # 显示db_workshop数据库的创建语句

use db_workshop; # 使用db_workshop数据库

show variables like "character_set_database"; # 字符集 utf8mb4
show variables like "collation_database"; # 排序方式 utf8mb4_general_ci 不区分大小写

create table table_people (
    id INT AUTO_INCREMENT PRIMARY KEY,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    date_of_birth DATE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

show tables from db_workshop; # 显示所有表
```
